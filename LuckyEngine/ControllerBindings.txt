# Xbox controller bindings for Lucky's camera (X500View). Read when Play starts: edit, save, press Play again.
# Flying (sticks, arm, modes, gripper) is the ground station's: src/ControllerBindings.txt.
# Fixed keys: R resets the scene, C steps through the cameras (X500 orbit, follow, gripper), Q/E zoom, right mouse drag orbits the camera.
#
# Numbers are axes (as Lucky reports this Xbox controller): 0 left stick X, 1 left stick Y, 2 right stick X,
#   3 right stick Y, 4 left trigger, 5 right trigger. Triggers rest at -1.
#   A minus flips the direction (pushing a stick up or left reads negative), e.g. -1.
#   Where an action is on/off (zoom), an axis counts when pushed past halfway.
# bN is raw button number N. This controller (the newer Xbox Wireless Controller, with the Share button, over Bluetooth):
#   b0 A, b1 B, b3 X, b4 Y, b6 LB, b7 RB, b10 View, b11 Menu, b14 right stick click;
#   Share also arrives as b0 (checked in Lucky's log): Lucky can't tell it from A, so they always do the same thing.
#   the D-pad comes after the buttons: b15 up, b16 right, b17 down, b18 left.

deadzone     = 0.15          # stick values below this count as 0

# Camera
camera_yaw   = -2            # right stick left/right: orbit around
camera_pitch = -3            # right stick up/down: orbit up/down
zoom_in      = b15           # D-pad up
zoom_out     = b17           # D-pad down
camera_view  = b3            # X: next camera: X500 orbit -> follow (behind its tail) -> gripper (C on the keyboard)
